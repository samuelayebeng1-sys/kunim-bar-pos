// CART
function addToCart(id){
  const item=menuItems.find(i=>i.id===id);if(!item||item.stock<=0)return;
  const ex=cart.find(c=>c.id===id);
  if(ex){if(ex.qty>=item.stock)return;ex.qty++;}
  else cart.push({id:item.id,name:item.name,price:item.price,qty:1,photo:item.photo||''});
  renderCart();
}
function updQty(id,d){
  const idx=cart.findIndex(c=>c.id===id);if(idx<0)return;
  cart[idx].qty+=d;if(cart[idx].qty<=0)cart.splice(idx,1);
  renderCart();
}
